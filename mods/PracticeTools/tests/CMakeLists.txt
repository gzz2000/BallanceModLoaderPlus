# A native Ballanced test parent must supply the matching CK2, CK2_3D and
# VxMath targets (built in-tree or imported with their interface includes).
# This fixture uses renderer internals; it cannot link against retail SDK stubs.
foreach(runtime IN ITEMS CK2 CK2_3D VxMath)
    if(NOT TARGET ${runtime})
        message(FATAL_ERROR "PracticeTools timeline tests require ${runtime}")
    endif()
endforeach()
add_executable(practice_timeline_objects_test test_timeline_objects.cpp)
target_compile_features(practice_timeline_objects_test PRIVATE cxx_std_17)
target_link_libraries(practice_timeline_objects_test PRIVATE CK2_3D CK2 VxMath)
add_test(NAME practice_timeline_objects_test COMMAND practice_timeline_objects_test)
