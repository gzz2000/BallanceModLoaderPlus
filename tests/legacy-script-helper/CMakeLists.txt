cmake_minimum_required(VERSION 3.25)
project(LegacyScriptHelperRegression LANGUAGES CXX)

set(CK2_SOURCE_DIR "" CACHE PATH "CK2 source directory")
set(VXMATH_SOURCE_DIR "${CK2_SOURCE_DIR}/../VxMath" CACHE PATH "VxMath source directory")
set(BML_LIBRARY "" CACHE FILEPATH "Built BML library or import library")
set(CK2_LIBRARY "" CACHE FILEPATH "Built CK2 library or import library")
set(VXMATH_LIBRARY "" CACHE FILEPATH "Built VxMath library or import library")
foreach(LIBRARY IN ITEMS BML_LIBRARY CK2_LIBRARY VXMATH_LIBRARY)
    if(NOT EXISTS "${${LIBRARY}}")
        message(FATAL_ERROR "Set ${LIBRARY} to a built library")
    endif()
    get_filename_component(LIBRARY_DIR "${${LIBRARY}}" DIRECTORY)
    list(APPEND RUNTIME_DIRS "${LIBRARY_DIR}")
endforeach()
get_filename_component(BML_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
add_executable(LegacyScriptHelperRegression "${BML_ROOT}/tests/ck2/LegacyScriptHelperRegression.cpp")
target_compile_features(LegacyScriptHelperRegression PRIVATE cxx_std_20)
target_include_directories(LegacyScriptHelperRegression PRIVATE
    "${BML_ROOT}/include" "${CK2_SOURCE_DIR}/include" "${VXMATH_SOURCE_DIR}/include")
target_link_libraries(LegacyScriptHelperRegression PRIVATE
    "${BML_LIBRARY}" "${CK2_LIBRARY}" "${VXMATH_LIBRARY}")
set_target_properties(LegacyScriptHelperRegression PROPERTIES BUILD_RPATH "${RUNTIME_DIRS}")
enable_testing()
add_test(NAME LegacyScriptHelperContexts COMMAND LegacyScriptHelperRegression contexts)
add_test(NAME LegacyScriptHelperReload COMMAND LegacyScriptHelperRegression reload)
